Options -Indexes
DirectoryIndex index.html

<IfModule mod_rewrite.c>
    RewriteEngine On

    # Preserve bearer tokens used by the student and administrator portals.
    RewriteCond %{HTTP:Authorization} .
    RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

    # Run Laravel for same-domain API requests. The END flag prevents the
    # internal backend path from being processed as a public request.
    RewriteRule ^api(?:/.*)?$ backend/public/index.php [END,QSA]

    # Never expose application source, deployment files, or Git metadata
    # through the main website document root.
    RewriteRule ^(?:backend|frontend|deployment|screenshots|\.git)(?:/|$) - [F,L,NC]

    # Block repository documentation and local tooling files at the root.
    RewriteRule ^(?:README\.md|PHP-COMPOSER-INSTALL-GUIDE\.txt|\.gitignore)$ - [F,L,NC]
</IfModule>

<FilesMatch "^\.">
    Require all denied
</FilesMatch>

<IfModule mod_headers.c>
    <FilesMatch "\.(?:css|js|png|jpg|jpeg|gif|svg|webp|ico|woff|woff2)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>
</IfModule>
